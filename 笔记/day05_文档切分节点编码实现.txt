04—掌柜智库_文档切片节点_切分与合并章节的框架搭建_编码实现
    def _split_long_merge_short(self, sections: List[dict], max_content_length: int = 1000,
                                min_content_length: int = 100) -> List[dict]:
        """
        长段落进行二次切分，短段落进行合并
        :param sections: 按标题切分的段落列表
        :param max_content_length:  每个section段落的最大长度
        :param min_content_length:
        :return: 重新切分合并后的结果列表
        """
        current_sections = []
        # 1.长切分
        for section in sections:
            # 收集切分和没切分的段落
            current_sections.extend(self._split_long_section(section, max_content_length))

        # 2.短合并
        final_sections = self._merge_short_section(current_sections, min_content_length)

        # 3.返回
        return final_sections

09—掌柜智库_文档切片节点_组装最终的chunks_编码实现
    # =======================================================================================
    #                #4.组装切片
    # =======================================================================================
    def _assemble_chunk(self, final_chunks: List[dict]) -> List[dict]:
        self.log_step("step4", "组装切片")

        chunks = []

        # 4.1 title + body
        for chunk in final_chunks:
            assemble_chunk = {
                "title": chunk.get("title"),
                "parent_title": chunk.get("parent_title"),
                "file_title": chunk.get("file_title"),
                "content": chunk.get("title") + "\n\n" + chunk.get("body")
            }
            chunks.append(assemble_chunk)

        # 4.2 构建最终 chunk  切片
        return chunks

    def _log_summary(self, raw_content: str, chunks: List[dict], max_length: int):
        """输出切分统计信息"""
        self.log_step("step5", "输出统计")

        lines_count = raw_content.count("\n") + 1
        self.logger.info(f"原文档行数: {lines_count}")
        self.logger.info(f"最终切分章节数: {len(chunks)}")
        self.logger.info(f"最大切片长度: {max_length}")

        if chunks:
            self.logger.info("章节预览:")
            for i, sec in enumerate(chunks[:5]):
                title = sec.get("title", "")[:30]
                self.logger.info(f"  {i + 1}. {title}...")
            if len(chunks) > 5:
                self.logger.info(f"  ... 还有 {len(chunks) - 5} 个章节")

    def _backup_chunks(self, state: ImportGraphState, sections: List[dict]):
        """将切分结果备份到 JSON 文件"""
        self.log_step("step6", "备份切片")

        local_dir = state.get("file_dir", "")
        if not local_dir:
            self.logger.debug("未设置 file_dir，跳过备份")
            return

        try:
            os.makedirs(local_dir, exist_ok=True)
            output_path = os.path.join(local_dir, "chunks.json")
            with open(output_path, "w", encoding="utf-8") as f:
                json.dump(sections, f, ensure_ascii=False, indent=2)
            self.logger.info(f"已备份到: {output_path}")
        except Exception as e:
            self.logger.warning(f"备份失败: {e}")
